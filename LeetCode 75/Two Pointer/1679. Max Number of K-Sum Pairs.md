# Complexity
- Time complexity:
  - O(nlogn + n)
  - o(n)

- Space complexity:
  - o(1)
  - O(n)

function maxOperations(nums: number[], k: number): number {
    let ops = 0
    nums.sort((a, b) => a - b)
    let l = 0
    let r = nums.length - 1;

    while (l < r) {
        if ((nums[l] + nums[r]) > k) {
            r--
        } else if ((nums[l] + nums[r]) < k) {
            l++
        } else {
            ops++
            l++
            r--
        }
    }

    return ops
}
